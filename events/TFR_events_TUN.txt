#No-Confidence Vote
country_event = {
	id = tunisia.2
	title = tunisia.2.t
	desc = tunisia.2.d
	picture = GFX_Tunisia_coup_report
	
	fire_only_once = yes
	is_triggered_only = yes
	
	trigger = {
		tag = TUN
	}

	immediate = {
		hidden_effect = {
			if = {
				limit = {
					has_game_rule = {
						rule = TFRNS_Maghreb_Mauritania_Pressure
						option = TFRNS_Maghreb_Mauritania_Pressure_Yes
					}
				}
				set_global_flag = MRT_algeria_pressured_global
			}
			else_if = {
				limit = {
					has_game_rule = {
						rule = TFRNS_Maghreb_Mauritania_Pressure
						option = TFRNS_Maghreb_Mauritania_Pressure_No
					}
				}
			}
			else = {
				random_list = {
					60 = {
						set_global_flag = MRT_algeria_pressured_global
					}
					40 = {
					}
				}
			}
		}
	}
	
	option = { #Success
		name = tunisia.2.a
		ai_chance = {
			base = 1
			modifier = {
				factor = 0
				has_game_rule = {
					rule = TFRNS_Tunisia_2020_crisis
					option = TFRNS_Tunisia_2020_crisis_2
				}
			}
		}
		country_event = {
			id = tunisia.3
			days = 57
		}
	}
	option = { #Failure
		name = tunisia.2.b
		ai_chance = {
			base = 1
			modifier = {
				factor = 0
				has_game_rule = {
					rule = TFRNS_Tunisia_2020_crisis
					option = TFRNS_Tunisia_2020_crisis_1
				}
			}
		}
		country_event = {
			id = tunisia.1
			days = 148
		}
	}
}
