					set_country_flag = TFRNS_HRL_Haftar_Invasion_Participant
					set_country_flag = TFRNS_Source_HRL_Haftar

				FROM = {
					set_country_flag = TFRNS_HRL_Haftar_Invasion_Participant
					TFRNS_remove_mushroom_buff_if_present = yes
				}
				if = {
					limit = {
						has_global_flag = Algerian_Civil_War_Flag
						FROM = { OR = { tag = ALG tag = AG1 tag = AG2 tag = AG3 } }

							rule = TFRNS_Algeria_Civil_War_Winner
							option = TFRNS_Algeria_Civil_War_Winner_ALG
						}
					}
					set_global_flag = TFRNS_HRL_Algerian_CW_Buff_Suppressed
					ALG = { TFRNS_remove_mushroom_buff_if_present = yes }

					limit = {

		effect = {
			# Reconcile the invasion buffs after country splits and delayed war starts.

				limit = {
					HRL = {
						has_war = yes
						has_global_flag = TFRNS_HRL_Haftar_Invasion_Active

							rule = TFRNS_HRL_Haftar_Invasion_Outcome
							option = TFRNS_HRL_Haftar_Invasion_Outcome_HRL_Wins
						}
					}
				}
				HRL = {
					set_country_flag = TFRNS_HRL_Haftar_Invasion_Participant
					set_country_flag = TFRNS_Source_HRL_Haftar
					TFRNS_apply_attack_mushroom_buff_if_missing = yes
				}
				every_country = {
					limit = { has_war_with = HRL }
					set_country_flag = TFRNS_HRL_Haftar_Invasion_Participant

						}
					}

					limit = { has_war_with = HRL }
					set_country_flag = TFRNS_HRL_Gaddafi_Invasion_Participant
					set_country_flag = TFRNS_Source_HRL_Gaddafi
					TFRNS_apply_defense_mushroom_buff_if_missing = yes
				}
			}

				limit = {
					tag = HRL
					has_global_flag = TFRNS_HRL_Haftar_Morocco_Final_Stage

						any_country = {
							limit = {
								has_country_flag = TFRNS_HRL_Haftar_Invasion_Participant
								has_war_with = HRL
							}
						}
					}
				}
				clr_global_flag = TFRNS_HRL_Haftar_Invasion_Active
				clr_global_flag = TFRNS_HRL_Haftar_Morocco_Final_Stage
				every_country = {
					limit = { has_country_flag = TFRNS_HRL_Haftar_Invasion_Participant }
					clr_country_flag = TFRNS_HRL_Haftar_Invasion_Participant
					clr_country_flag = TFRNS_Source_HRL_Haftar
					TFRNS_remove_mushroom_buff_if_no_other_campaign_owns_it = yes
				}
			}
			if = {
				limit = {
					tag = HRL
					has_global_flag = TFRNS_HRL_Gaddafi_Morocco_Final_Stage
					NOT = {
						any_country = {
							limit = {
								has_country_flag = TFRNS_HRL_Gaddafi_Invasion_Participant

					}
