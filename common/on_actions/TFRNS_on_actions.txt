on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = {
					has_game_rule = {
						rule = TFRNS_Total_War
						option = TFRNS_Total_War_ON
					}
				}
				every_possible_country = {
					add_ideas = TFRNS_no_surrender
				}
			}
		}
	}
	on_weekly = {
		effect = {
			if = {
				limit = {
					has_game_rule = {
						rule = TFRNS_Total_War
						option = TFRNS_Total_War_ON
					}
					NOT = {
						has_idea = TFRNS_no_surrender
					}
				}
				add_ideas = TFRNS_no_surrender
			}
		}
	}
}
