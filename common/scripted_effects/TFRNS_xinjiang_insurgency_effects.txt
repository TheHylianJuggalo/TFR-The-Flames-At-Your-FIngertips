# TFR No Surrender - Xinjiang Insurgency winner enforcement
#
# Keeps the Xinjiang Insurgency winner rule honest: while the insurgency is
# being fought only the rule-selected side may hold the Mushroom Army buff, and
# every trace is released once the war ends. Safe to call every week - the
# effect is fully idempotent.

# --- release helper (current scope = the country losing the buff) ---------

TFRNS_release_xinjiang_insurgency_buff = {
	if = {
		limit = { has_country_flag = TFRNS_Source_Xinjiang_Insurgency }
		clr_country_flag = TFRNS_Source_Xinjiang_Insurgency
		TFRNS_remove_mushroom_buff_if_no_other_campaign_owns_it = yes
	}
}

# --- reconciliation -------------------------------------------------------

# Xinjiang Insurgency winner: XIN (Construction Corps) or the East Turkestan
# rebels (SIK, HUT, ETL).
TFRNS_reconcile_xinjiang_insurgency_buff = {
	if = {
		limit = {
			TFRNS_xinjiang_insurgency_ongoing = yes
			has_game_rule = {
				rule = TFRNS_Xinjiang_Insurgency
				option = TFRNS_Xinjiang_Insurgency_CRUSHED
			}
		}
		# Winner: the Xinjiang Production and Construction Corps.
		if = {
			limit = { country_exists = XIN }
			XIN = {
				set_country_flag = TFRNS_Source_Xinjiang_Insurgency
				TFRNS_apply_attack_mushroom_buff_if_missing = yes
			}
		}
		if = { limit = { country_exists = SIK } SIK = { TFRNS_release_xinjiang_insurgency_buff = yes } }
		if = { limit = { country_exists = HUT } HUT = { TFRNS_release_xinjiang_insurgency_buff = yes } }
		if = { limit = { country_exists = ETL } ETL = { TFRNS_release_xinjiang_insurgency_buff = yes } }
	}
	else_if = {
		limit = {
			TFRNS_xinjiang_insurgency_ongoing = yes
			has_game_rule = {
				rule = TFRNS_Xinjiang_Insurgency
				option = TFRNS_Xinjiang_Insurgency_UNCONTROLLED
			}
		}
		# Winner: the East Turkestan rebels.
		if = {
			limit = { country_exists = SIK }
			SIK = {
				set_country_flag = TFRNS_Source_Xinjiang_Insurgency
				TFRNS_apply_attack_mushroom_buff_if_missing = yes
			}
		}
		if = {
			limit = { country_exists = HUT }
			HUT = {
				set_country_flag = TFRNS_Source_Xinjiang_Insurgency
				TFRNS_apply_attack_mushroom_buff_if_missing = yes
			}
		}
		if = {
			limit = { country_exists = ETL }
			ETL = {
				set_country_flag = TFRNS_Source_Xinjiang_Insurgency
				TFRNS_apply_attack_mushroom_buff_if_missing = yes
			}
		}
		if = { limit = { country_exists = XIN } XIN = { TFRNS_release_xinjiang_insurgency_buff = yes } }
	}
	else = {
		# DEFAULT, or the insurgency is over: nobody keeps the buff.
		if = { limit = { country_exists = XIN } XIN = { TFRNS_release_xinjiang_insurgency_buff = yes } }
		if = { limit = { country_exists = SIK } SIK = { TFRNS_release_xinjiang_insurgency_buff = yes } }
		if = { limit = { country_exists = HUT } HUT = { TFRNS_release_xinjiang_insurgency_buff = yes } }
		if = { limit = { country_exists = ETL } ETL = { TFRNS_release_xinjiang_insurgency_buff = yes } }
	}
}
